entity andor is
   port ( i1, i2, i3 : in bit;
          output : out bit
          );
   end andor;
 
   architecture struct of andor is
     -- zakłada się że gdzieś już została zdefiniowana bramka AND2
   component AND2 is
   port( A, B : in bit;
         F1 : out bit
         );
   end component;
 
    -- zakłada się że gdzieś już została zdefiniowana bramka OR2
    component OR2 is
    port( X, Y : in bit;
          F2 : out bit
         );
    end component;
 
    --deklaracja pomocniczego sygnału, wykorzystywanego do łączenia komponentów ze sobą
    signal write : bit;
 
    begin
         Gate1 : AND2 port map (A => i1, B => i2, F1 => write);
         Gate2 : OR2  port map (write, i3, output);
    end struct;
